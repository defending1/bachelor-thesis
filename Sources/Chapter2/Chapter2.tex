\chapter{Unicità della fattorizzazione CP}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

Nel 1977 Kruskal dimostra l'unicità della fattorizzazione CP nel caso $ 3- $dimensionale con un argomento di 40 pagine
\cite{KRUSKAL197795}, di non semplice lettura. La sua idea è stata riadattata da diversi matematici, e attualmente abbiamo
due dimostrazioni concise seguendo due strade diverse, che entrambe sfruttano il permutation lemma
formulato da Kruskal. La prima, scritta da Stegeman e
Sidiropoulos~\cite{STEGEMAN2007540} di 16 pagine, e poi riadattata da
Landsberg~\cite{landsberg2011tensors} in un argomento geometrico di sole 3 pagine. La seconda di Rhodes (circa 9
pagine) in \cite{rhodes2009conciseproofkruskalstheorem} in coordinate. Il caso generale, dimostrato
da Sidiropoulos e Bro~\cite{sidiropoulos2000uniqueness} si ottiene (in maniera non ovvia)
riconducendosi al caso $ 3-$dimensionale, pertanto riteniamo più interessante trattare solamente il caso base.

\begin{definition}[Essenziale unicità CP]
  Una fattorizzazione CP di rango $ r $ di un tensore $ \Tn{x} \dsqr{U_1, \ldots, U_d}$ è
  \emph{essenzialmente unica} se, per ogni altra fattorizzazione $ \Tn{x} = \dsqr{\tilde{U}_1,
  \ldots, \tilde{U}_d}$ di rango $ r $, esiste una patrice di permutazione $ P \in \R^{r \times r} $
  e delle matrici diagonali $ D_1, \ldots, D_d \in \R^{r \times r} $ che soddisfano $ D_1 D_2 \cdots D_d =
  I$ e $U_j = \tilde{U_j}D_jP$, per ogni $ j = 1, \ldots, d $.
\end{definition}

\begin{definition}[$ k- $rango]
  Data $ A \in \R^{m\times n} $, il suo \emph{$ K- $rango} è il massimo $ K \in \N $ tale che ogni
  scelta di $ K $ colonne in $ A $ comprende colonne linearmente indipendenti. Lo denoteremo con il
  simbolo $ \krk{A} $.
\end{definition}
\begin{remark}
  Dalla definizione segue che $ \krk{A} \le \rk(A) $ per ogni matrice $ A \in \R^{m \times n} $. In
  particolare, se $ A $ è full-rank, $ \krk{A} = \rk(A) $; se invece $ A $ ha due colonne linearmente
  dipendenti, $ \krk{A} \le 1 $.
\end{remark}

\begin{theorem}[Kruskal 3D, Sidiropoulos e Bro]
  Una fattorizzazione CP di un tensore $ \Tn{X} = \dsqr{U_1, \ldots, U_d}$ di rango $ r $ è essenzialmente unica se
  \begin{equation*}
    \sum_{j=1}^d \krk{U_j} \ge 2r + d -1.
  \end{equation*}
\end{theorem}

Per il caso $ d=4 $ introduciamo il seguente lemma:

\begin{lemma}[$K-$rango del prodotto di Khatri-Rao]
  Consideriamo $ A = \begin{bmatrix} a_1 \mid \cdots \mid a_r \end{bmatrix} \in \C^{m\times p} $ e $ B =
  \begin{bmatrix} b_1 \mid \cdots \mid b_r \end{bmatrix} \in \C^{n \times p}$. Allora, se $ \krk{A}
  \ge  1$ e $ \krk{B} \ge 1 $ vale che
  \begin{equation*}
    \krk{A \krp B} \ge \min(\krk{A} + \krk{B} - 1, r),
  \end{equation*}
  mentre se almeno una tra $ A $ o $ B $ ha $ K- $rango nullo, vale
  \begin{equation*}
    \krk{A \krp B} = 0.
  \end{equation*}
\end{lemma}
\begin{proof}                           
  Assumiamo a meno di scambiare le due matrici che $ \krk{A} $ sia nullo. Ciò equivale a dire che $
  A $ contiene almeno una colonna nulla, da cui per la Definizione \ref{def:krn} il prodotto $ A
  \krp B$ avrà almeno una colonna nulla, da cui $ \krk{A \krp B} $.
  
  Mostriamo la prima stima procedendo per assurdo. Consideriamo l'insieme
  \begin{equation*}
    S = \inf \left\{ s \in \N \bigm| b_{i_1} \krn a_{i_1}, \ldots, b_{i_s} \krn a_{i_s} \text{ sono
    colonne linearmente indipendenti di } A \krp B \right\}.
  \end{equation*}
  Se $ S = -\infty $, ossia non è possibile estrarre alcuna collezione di colonne linearmente
  indipendenti da $ A \krp B $, poniamo $ S = d+1 $ e $ \krk{B \krp A} = S-1 $.

La condizione di Kruskal è sufficiente ma non necessaria. Il seguente risultato fornisce una
condizione necessaria ma non sufficiente:

\begin{theorem}
  Dato un tensore $ \Tn{x} = \dsqr{U_1, \ldots, U_d}$ tale per cui $ \min_j \left\{ \rk(U_d \krp
  \cdots \krp U_{j+1} \krp U_{j-1} \krp \cdots \krp U_1) \right\} < r$, allora la fattorizzazine $ \Tn{x} =
  \dsqr{U_1, \ldots, U_d} $ non è essenzialmente unica.
\end{theorem}
